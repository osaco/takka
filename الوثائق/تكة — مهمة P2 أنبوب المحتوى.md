# 🟡 P2 — أنبوب المحتوى (Content Pipeline)
> **المدة:** أسبوع  
> **البوابة:** P1 مفتوحة  
> **القاعدة:** Zod schema + مدقّق + حزمة JSON تجريبية

---

## 🎯 الأهداف

1. بناء Zod schemas لجميع أنواع المحتوى (Skill, Activity, Pack, Evidence)
2. بناء مدقّق محتوى آلي (validator.ts) يفحص: صحة، لغة، تشكيل، عمر، تلميح
3. بناء محمّل حزم (loader.ts) يحمّل JSON + موارد مرتبطة
4. إنشاء حزمة تجريبية للصف الأول (g1-ar-v0.1) للتجربة
5. التأكد من أن النشاط بدون `source` أو `audit.status` يرفض البناء

### Prompt 2 — المدقّق التربوي الآلي

```
أنشئ مدقّق محتوى آلي في src/domain/content/validator.ts

المطلوب:
- فحص صحة البيانات (schema validation)
- فحص اللغة العربية (تشكيل، إملاء)
- فحص العمر المناسب (age-appropriateness)
- فحص التلميحات (hint quality)
- فحص المصادر (source validation)
- رفض أي نشاط بدون source أو audit.status = 'internal_reviewed'

```typescript
import { z } from 'zod'
import type { Activity, Skill, ContentPack } from './schema'

export type AuditResult = 
  | { ok: true; activityId: string; warnings: string[] }
  | { ok: false; activityId: string; errors: string[] }

export class ContentValidator {
  /** فحص نشاط واحد */
  validateActivity(activity: Activity): AuditResult {
    const errors: string[] = []
    const warnings: string[] = []

    // 1. Source validation
    if (!activity.source || activity.source.kind === 'foundational') {
      warnings.push('نشاط بدون مصدر موثّق (foundational)')
    }
    if (activity.source?.file && !activity.source.file.endsWith('.pdf')) {
      warnings.push('ملف المصدر ليس PDF')
    }

    // 2. Audit status
    if (activity.audit.status !== 'internal_reviewed') {
      errors.push('النشاط ليس لديه فحص داخلي (internal_reviewed)')
      return { ok: false, activityId: activity.id, errors }
    }

    // 3. Title length
    if (activity.title.length > 100) {
      warnings.push('عنوان النشاط طويل جداً (>100 حرف)')
    }

    // 4. Config validation per template
    if (activity.template === 'drag_match') {
      const pairs = (activity.config as any)?.pairs
      if (!pairs || !Array.isArray(pairs) || pairs.length === 0) {
        errors.push('قالب سحب+مطابقة يحتاج pairs')
      }
      if (pairs) {
        for (const pair of pairs) {
          if (!pair.draggable?.id || !pair.target?.id) {
            errors.push('pair يفتقد draggable.id أو target.id')
          }
        }
      }
    }

    if (activity.template === 'listen_identify') {
      const audioFile = (activity.config as any)?.audioFile
      if (!audioFile) {
        warnings.push('قالب استماع بدون audioFile (سيعتمد على النص فقط)')
      }
      const options = (activity.config as any)?.options
      if (!options || !Array.isArray(options) || options.length < 2) {
        errors.push('قالب استماع يحتاج ≥2 خيارات')
      }
    }

    if (activity.template === 'trace_path') {
      const strokePath = (activity.config as any)?.strokePath
      if (!strokePath) {
        errors.push('قالب تتبع يحتاج strokePath SVG')
      }
    }

    // 5. Feedback quality
    if (!activity.feedback?.success || activity.feedback.success.length < 5) {
      warnings.push('رسالة النجاح قصيرة جداً')
    }
    if (!activity.feedback?.retry || activity.feedback.retry.length < 5) {
      warnings.push('رسالة重试 قصيرة جداً')
    }

    // 6. Hint quality (if level > 1)
    if (activity.scaffolding.level > 1 && !activity.scaffolding.hint) {
      warnings.push('سقالة مستوى > 1 بدون تلميح نصي')
    }

    // 7. Arabic text validation
    const hasArabic = activity.title.includes(/[\u0600-\u06FF]/) || 
                      Object.values(activity.config).some(v => typeof v === 'string' && /[.\u0600-\u06FF]/.test(v))
    if (hasArabic && !this.checkArabicTashkeel(activity)) {
      warnings.push('نص عربي بدون تشكيل قد يصعّب القراءة')
    }

    if (errors.length > 0) {
      return { ok: false, activityId: activity.id, errors }
    }

    return { ok: true, activityId: activity.id, warnings }
  }

  /** فحص تشكيل عربي أساسي */
  private checkArabicTashkeel(activity: Activity): boolean {
    const arabicText = activity.title + ' ' + 
      Object.values(activity.config).filter(v => typeof v === 'string').join(' ')
    // إذا كان النص العربي يحتوي على حركات، نعتبره مقبولاً
    return /[\u064B-\u065F]/.test(arabicText) || arabicText.length < 10
  }

  /** فحص حزمة كاملة */
  validatePack(pack: ContentPack): { valid: number; invalid: number; results: AuditResult[] } {
    const results: AuditResult[] = []
    let valid = 0
    let invalid = 0

    for (const activity of pack.activities) {
      const result = this.validateActivity(activity)
      results.push(result)
      if (result.ok) valid++
      else invalid++
    }

    return { valid, invalid, results }
  }

  /** فحص مهارات */
  validateSkills(skills: Skill[]): { valid: number; invalid: number; errors: string[] } {
    const errors: string[] = []
    let valid = 0

    for (const skill of skills) {
      // التحقق من عدم تكرار المعرف
      if (skills.filter(s => s.id === skill.id).length > 1) {
        errors.push(`مهارة مكررة: ${skill.id}`)
        continue
      }
      // التحقق من prerequisites
      for (const prereq of skill.prerequisites) {
        if (!skills.find(s => s.id === prereq)) {
          errors.push(`مهارة ${skill.id} تحتاج prerequisite غير موجودة: ${prereq}`)
        }
      }
      valid++
    }

    return { valid, invalid: errors.length, errors }
  }
}

export const validator = new ContentValidator()

```

```typescript
import type { ContentPack } from './schema'
import { ContentPackSchema, ActivitySchema } from './schema'
import { validator } from './validator'

export interface LoadResult {
  success: boolean
  pack?: ContentPack
  errors?: string[]
  warnings?: string[]
}

export class ContentLoader {
  /** تحميل حزمة من JSON string */
  async loadFromJson(json: string): Promise<LoadResult> {
    try {
      const data = JSON.parse(json)
      
      // Validate schema
      const parsed = ContentPackSchema.safeParse(data)
      if (!parsed.success) {
        return {
          success: false,
          errors: parsed.error.issues.map(i => `${i.path.join('.')}: ${i.message}`),
        }
      }

      const pack = parsed.data
      
      // Validate content quality
      const validation = validator.validatePack(pack)
      const warnings = validation.results
        .filter(r => !r.ok === false)
        .flatMap(r => r.ok ? r.warnings : [])
      
      const errors = validation.results
        .filter(r => !r.ok)
        .flatMap(r => r.errors)

      if (errors.length > 0) {
        return { success: false, errors }
      }

      return { success: true, pack, warnings }
    } catch (e) {
      return {
        success: false,
        errors: [`فشل تحليل JSON: ${e instanceof Error ? e.message : String(e)}`],
      }
    }
  }

  /** تحميل حزمة من ملف محلي */
  async loadFromFile(path: string): Promise<LoadResult> {
    try {
      const response = await fetch(path)
      const json = await response.text()
      return this.loadFromJson(json)
    } catch (e) {
      return {
        success: false,
        errors: [`فشل تحميل الملف: ${e instanceof Error ? e.message : String(e)}`],
      }
    }
  }

  /** تحميل جميع الحزم من مجلد */
  async loadAllPacks(basePath: string): Promise<LoadResult[]> {
    // في الوضع الحقيقي نقرأ من文件系统 أو من content-packs/
    // هنا نعيّدها كـ mock للاختبارات
    return []
  }

  /** تصدير حزمة إلى JSON */
  exportPack(pack: ContentPack): string {
    return JSON.stringify(pack, null, 2)
  }

  /** حساب تغطية المنهاج */
  calculateCoverage(pack: ContentPack, totalExpectedSkills: number): { covered: number; percentage: number } {
    const covered = pack.skills.length
    return {
      covered,
      percentage: Math.round((covered / totalExpectedSkills) * 100),
    }
  }
}

export const contentLoader = new ContentLoader()
