### state-machine.ts
```typescript
import type { Evidence, EvidenceKind, SkillState } from '../types'

/** قاعدة الإتقان: مهارة متقنة = 3 أدلة مستقلة في جلستين مختلفتين بفاصل ≥ يوم */
const INDEPENDENT_THRESHOLD = 3
const MIN_SESSIONS = 2
const MIN_GAP_DAYS = 1

export type Transition = 
  | 'not_reached→tried'
  | 'tried→with_help'
  | 'with_help→independent'
  | 'independent→decay_review'
  | 'independent→not_reached'

export function computeNewState(
  currentState: SkillState,
  evidence: Evidence,
  previousEvidences: Evidence[],
): SkillState {
  // إذا حصل على دليل مستقل
  if (evidence.kind === 'independent') {
    const independentCount = previousEvidences.filter(e => e.kind === 'independent').length + 1
    const uniqueSessions = new Set(previousEvidences.map(e => e.itemId)).size + 1
    const lastIndependent = previousEvidences.filter(e => e.kind === 'independent').pop()
    const gapDays = lastIndependent 
      ? (evidence.at - lastIndependent.at) / (1000 * 60 * 60 * 24)
      : 0
    
    if (independentCount >= INDEPENDENT_THRESHOLD 
        && uniqueSessions >= MIN_SESSIONS 
        && gapDays >= MIN_GAP_DAYS) {
      return 'independent'
    }
  }
  
  // إذا جرب ولم يُنجح
  if (evidence.kind === 'exposed_only' || evidence.attempts > 0) {
    if (currentState === 'not_reached') return 'tried'
  }
  
  // إذا نجح بمساعدة
  if (evidence.kind === 'with_hint' || evidence.kind === 'with_scaffold') {
    if (currentState === 'tried' || currentState === 'not_reached') return 'with_help'
  }
  
  return currentState
}

export function countIndependentEvidence(evidences: Evidence[]): number {
  return evidences.filter(e => e.kind === 'independent').length
}

export function getLastIndependentAt(evidences: Evidence[]): number | null {
  const independents = evidences.filter(e => e.kind === 'independent')
  return independents.length > 0 ? independents[independents.length - 1].at : null
}
```

### evidence-store.ts
```typescript
import type { Evidence, SkillState, ChildProfile } from '../types'
import { computeNewState } from './state-machine'

export class EvidenceStore {
  private evidencesBySkill = new Map<string, Evidence[]>()
  private profiles = new Map<string, ChildProfile>()

  addEvidence(profileId: string, evidence: Evidence): void {
    if (!this.evidencesBySkill.has(evidence.skillId)) {
      this.evidencesBySkill.set(evidence.skillId, [])
    }
    this.evidencesBySkill.get(evidence.skillId)!.push(evidence)
  }

  getSkillState(profileId: string, skillId: string): SkillState {
    const evidences = this.evidencesBySkill.get(skillId) ?? []
    if (evidences.length === 0) return 'not_reached'
    
    let state: SkillState = 'not_reached'
    for (const evidence of evidences) {
      state = computeNewState(state, evidence, evidences.filter(e => e.at < evidence.at))
    }
    return state
  }

  getSkillResults(profileId: string): Map<string, SkillState> {
    const results = new Map<string, SkillState>()
    for (const [skillId] of this.evidencesBySkill) {
      results.set(skillId, this.getSkillState(profileId, skillId))
    }
    return results
  }

  hasReachedSkill(profileId: string, skillId: string): boolean {
    return this.getSkillState(profileId, skillId) === 'independent'
  }
}
```

### decay.ts
```typescript
import type { Evidence, SkillState } from '../types'

/** إذا مرّ 21 يوماً بلا دليل مستقل، تعود المهارة إلى "تحتاج مراجعة" للاقتراح فقط */
const DECAY_THRESHOLD_DAYS = 21

export function checkDecay(
  skillId: string,
  lastIndependentAt: number | null,
  currentStates: Record<string, SkillState>,
): { skillId: string; needsReview: boolean; reason: string } {
  if (lastIndependentAt === null) {
    return { skillId, needsReview: false, reason: 'never_tried' }
  }
  
  const daysSinceLastIndependent = (Date.now() - lastIndependentAt) / (1000 * 60 * 60 * 24)
  
  if (daysSinceLastIndependent > DECAY_THRESHOLD_DAYS && currentStates[skillId] === 'independent') {
    return { 
      skillId, 
      needsReview: true, 
      reason: `لم يُظهر إتقاناً مستقلاً منذ ${Math.round(daysSinceLastIndependent)} يوماً` 
    }
  }
  
  return { skillId, needsReview: false, reason: 'ok' }
}

export function shouldScheduleReview(
  skillId: string,
  states: Record<string, SkillState>,
  lastAttempts: Record<string, number>,
): boolean {
  const lastAttempt = lastAttempts[skillId]
  if (!lastAttempt) return false
  
  const daysSinceAttempt = (Date.now() - lastAttempt) / (1000 * 60 * 60 * 24)
  const state = states[skillId]
  
  // المهام "with_help" أو "tried" تحتاج مراجعة بعد 3 أيام
  if ((state === 'with_help' || state === 'tried') && daysSinceAttempt >= 3) {
    return true
  }
  
  // المهام "independent" تحتاج مراجعة بعد 21 يوم (decay)
  if (state === 'independent' && daysSinceAttempt > DECAY_THRESHOLD_DAYS) {
    return true
  }
  
  return false
}
```

```typescript
export interface BridgeOption {
  skillId: string
  prerequisiteSkillId: string
  title: string
  activityIds: string[]
}

export function findBridge(
  failedSkillId: string,
  allSkills: { id: string; prerequisites: string[] }[],
): BridgeOption | null {
  const skill = allSkills.find(s => s.id === failedSkillId)
  if (!skill || skill.prerequisites.length === 0) return null
  
  // نبحث عن مهارة أساسية غير متقنة
  for (const prereqId of skill.prerequisites) {
    // يمكن توسيع هذا ليتحقق من حالة الطفل
    return {
      skillId: failedSkillId,
      prerequisiteSkillId: prereqId,
      title: `هل تريد أن تعيد ${prereqId} أولاً؟`,
      activityIds: [],
    }
  }
  
  return null
}
```

```typescript
import type { Grade, SubjectId, Skill, Evidence, SkillState } from '../types'
import { shouldScheduleReview } from '../scaffolding/levels'

interface SessionConfig {
  grade: Grade
  subject: SubjectId
  maxSkills: number
  maxDurationMinutes: number
}

export interface Session {
  id: string
  skills: Skill[]
  evidence: Evidence[]
  startedAt: number
  config: SessionConfig
}

export function buildSession(
  skills: Skill[],
  childStates: Map<string, SkillState>,
  lastAttempts: Record<string, number>,
  config: SessionConfig,
): Session {
  const now = Date.now()
  
  // الأولوية: مهارات تحتاج مراجعة
  const reviewPriority = skills
    .filter(s => shouldScheduleReview(s.id, Object.fromEntries(childStates), lastAttempts))
    .slice(0, Math.floor(config.maxSkills * 0.4))
  
  // ثم مهارات غير متحققة
  const newSkills = skills
    .filter(s => {
      const state = childStates.get(s.id)
      return state === 'not_reached' || state === 'tried'
    })
    .slice(0, config.maxSkills - reviewPriority.length)
  
  const sessionSkills = [...reviewPriority, ...newSkills]
  
  return {
    id: `session-${now}`,
    skills: sessionSkills,
    evidence: [],
    startedAt: now,
    config,
  }
}

export function estimateSessionDuration(skills: Skill[]): number {
  // 4-6 دقائق للصفوف 1-2، 7-10 للـ 3-4، 11-15 للـ 5-6
  const baseMinutes = 5
  return baseMinutes + (skills.length * 1.5)
}
```

```typescript
  from: SessionPhase
  to: SessionPhase
  condition: string
}

export const SESSION_PHASES_ORDER: SessionPhase[] = [
  'activation',
  'modeling',
  'guided_practice',
  'fluency_round',
  'transfer',
  'independent_check',
  'feedback',
  'closure',
]

export function getNextPhase(current: SessionPhase): SessionPhase | null {
  const idx = SESSION_PHASES_ORDER.indexOf(current)
  if (idx === -1 || idx >= SESSION_PHASES_ORDER.length - 1) return null
  return SESSION_PHASES_ORDER[idx + 1]
}

export function getSessionDurationForGrade(grade: number): { min: number; max: number } {
  if (grade <= 2) return { min: 4, max: 6 }
```

import type { Evidence, Session, ChildProfile } from '../types'

export interface SessionSummary {
  skillResults: Map<string, 'success' | 'needs_review' | 'gave_up'>
  totalAttempts: number
  totalHints: number
  durationMs: number
  evidence: Evidence[]
}

export function endSession(
  session: Session,
  finalEvidence: Evidence[],
  profile: ChildProfile,
): SessionSummary {
  const results = new Map<string, 'success' | 'needs_review' | 'gave_up'>()
  
  for (const skill of session.skills) {
    const skillEvidence = finalEvidence.filter(e => e.skillId === skill.id)
    const independentCount = skillEvidence.filter(e => e.kind === 'independent').length
    
    if (independentCount >= 3) {
      results.set(skill.id, 'success')
    } else if (skillEvidence.length > 0) {
      results.set(skill.id, 'needs_review')
    } else {
      results.set(skill.id, 'gave_up')
    }
  }
  
  return {
    skillResults: results,
    totalAttempts: finalEvidence.reduce((sum, e) => sum + e.attempts, 0),
    totalHints: finalEvidence.filter(e => e.kind === 'with_hint' || e.kind === 'with_scaffold').length,
    durationMs: Date.now() - session.startedAt,
    evidence: finalEvidence,
  }
}
