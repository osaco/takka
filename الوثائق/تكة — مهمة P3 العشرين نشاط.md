# 🟠 P3 — العشرون نشاطاً الحقيقي
> **المدة:** أسبوع  
> **البوابة:** P2 مفتوحة  
> **القاعدة:** 3 قوالب + 20 نشاطاً موثّقاً

---

## 🎯 الأهداف

1. بناء قالب **سحب ومطابقة** (DragMatchGame)
2. بناء قالب **تتبع مسارات** (TracePathGame)
3. بناء قالب **استماع وتمييز** (ListenIdentifyGame)
4. إنشاء 20 نشاطاً حقيقياً (10 عربي + 5 إنجليزي + 5 رياضيات)
5. ربط الأنشطة بالقوالب وعرضها في العالم

import React, { useCallback, useRef, useState } from 'react'
import { useDragMatch } from './hooks'
import type { Activity } from '@domain/content/schema'

interface DragMatchProps {
  activity: Activity
  onResult: (success: boolean, attempts: number) => void
}

export const DragMatchGame: React.FC<DragMatchProps> = ({ activity, onResult }) => {
  const config = activity.config as {
    prompt: string
    pairs: Array<{
      draggable: { id: string; label: string; image?: string }
      target: { id: string; label: string; char?: string }
    }>
    shuffleTargets?: boolean
  }

  const {
    draggedId,
    matchedPairs,
    wrongAttempts,
    handleDragStart,
    handleDragOver,
    handleDrop,
    handleDragEnd,
  } = useDragMatch(config, activity.scaffolding.level)

  const allMatched = matchedPairs.length === config.pairs.length

  React.useEffect(() => {
    if (allMatched) {
      onResult(true, wrongAttempts)
    }
  }, [allMatched])

  return (
    <div className="drag-match-game" dir="rtl">
      <p className="prompt">{config.prompt}</p>
      
      <div className="targets-container">
        {config.shuffleTargets 
          ? [...config.pairs].sort(() => Math.random() - 0.5).map(pair => (
              <div
                key={pair.target.id}
                className={`drop-target ${matchedPairs.includes(pair.target.id) ? 'matched' : ''}`}
                onDragOver={(e) => handleDragOver(e, pair.target.id)}
                onDrop={(e) => handleDrop(e, pair.target.id)}
              >
                {pair.target.char && (
                  <span className="target-char">{pair.target.char}</span>
                )}
                {pair.target.label && (
                  <span className="target-label">{pair.target.label}</span>
                )}
              </div>
            ))
          : config.pairs.map(pair => (
              <div
                key={pair.target.id}
                className={`drop-target ${matchedPairs.includes(pair.target.id) ? 'matched' : ''}`}
                onDragOver={(e) => handleDragOver(e, pair.target.id)}
                onDrop={(e) => handleDrop(e, pair.target.id)}
              >
                {pair.target.char && (
                  <span className="target-char">{pair.target.char}</span>
                )}
                {pair.target.label && (
                  <span className="target-label">{pair.target.label}</span>
                )}
              </div>
            ))
        }
      </div>

      <div className="draggables-container">
        {config.pairs.map(pair => (
          <div
            key={pair.draggable.id}
            className={`drag-item ${matchedPairs.includes(pair.draggable.id) ? 'hidden' : ''}`}
            draggable={!matchedPairs.includes(pair.draggable.id)}
            onDragStart={(e) => handleDragStart(e, pair.draggable.id)}
            onDragEnd={handleDragEnd}
          >
            {pair.draggable.image ? (
              <img src={pair.draggable.image} alt={pair.draggable.label} />
            ) : (
              <span>{pair.draggable.label}</span>
            )}
          </div>
        ))}
      </div>

      {activity.scaffolding.hint && wrongAttempts > 0 && (
        <div className="hint-box">
          {activity.scaffolding.hint}
        </div>
      )}
    </div>
  )
}

hooks.ts:
```typescript
import { useState, useCallback } from 'react'

interface UseDragMatchReturn {
  draggedId: string | null
  matchedPairs: string[]
  wrongAttempts: number
  handleDragStart: (e: React.DragEvent, id: string) => void
  handleDragOver: (e: React.DragEvent, targetId: string) => void
  handleDrop: (e: React.DragEvent, targetId: string) => void
  handleDragEnd: () => void
}

export function useDragMatch(
  config: {
    pairs: Array<{ draggable: { id: string }; target: { id: string } }>
    shuffleTargets?: boolean
  },
  scaffoldLevel: number,
): UseDragMatchReturn {
  const [draggedId, setDraggedId] = useState<string | null>(null)
  const [matchedPairs, setMatchedPairs] = useState<string[]>([])
  const [wrongAttempts, setWrongAttempts] = useState(0)

  const handleDragStart = useCallback((e: React.DragEvent, id: string) => {
    setDraggedId(id)
    e.dataTransfer.effectAllowed = 'move'
  }, [])

  const handleDragOver = useCallback((e: React.DragEvent, _targetId: string) => {
    e.preventDefault()
    e.dataTransfer.dropEffect = 'move'
  }, [])

  const handleDrop = useCallback((e: React.DragEvent, targetId: string) => {
    e.preventDefault()
    
    if (!draggedId) return

    // تحقق من المطابقة
    const pair = config.pairs.find(
      p => p.draggable.id === draggedId && p.target.id === targetId
    )

    if (pair) {
      setMatchedPairs(prev => [...prev, draggedId])
      setDraggedId(null)
    } else {
      setWrongAttempts(prev => prev + 1)
      setDraggedId(null)
    }
  }, [draggedId, config.pairs])

  const handleDragEnd = useCallback(() => {
    setDraggedId(null)
  }, [])

  return {
    draggedId,
    matchedPairs,
    wrongAttempts,
    handleDragStart,
    handleDragOver,
    handleDrop,
    handleDragEnd,
  }
}

```tsx
import React, { useEffect, useState } from 'react'
import { useAppStore } from '../store/app-store'
import { DragMatchGame } from '../../templates/DragMatch/DragMatchGame'
import { TracePathGame } from '../../templates/TracePath/TracePathGame'
import { ListenIdentifyGame } from '../../templates/ListenIdentify/ListenIdentifyGame'
import type { Activity, Evidence } from '@domain/content/schema'

interface ActivityScreenProps {
  activity: Activity
  onComplete: (evidence: Evidence) => void
  onExit: () => void
}

export const ActivityScreen: React.FC<ActivityScreenProps> = ({ activity, onComplete, onExit }) => {
  const { isSoundEnabled } = useAppStore()
  const [isComplete, setIsComplete] = useState(false)
  const [attempts, setAttempts] = useState(0)

  const handleResult = (success: boolean, attemptCount: number) => {
    setAttempts(attemptCount)
    if (success && !isComplete) {
      setIsComplete(true)
      const evidence: Evidence = {
        skillId: activity.skillId,
        at: Date.now(),
        kind: attemptCount === 0 ? 'independent' : 'with_hint',
        templateId: activity.template,
        itemId: activity.id,
        attempts: attemptCount + 1,
        durationMs: 0,
      }
      onComplete(evidence)
    }
  }

  if (isComplete) {
    return (
      <div className="activity-complete" dir="rtl">
        <h2>🎉 أحسنت! أتقنت هذه المهارة</h2>
        <p>الجزيرة ازدهرت خطوة صغيرة</p>
        <button onClick={onExit} className="btn-primary">
          ارجع للخريطة
        </button>
      </div>
    )
  }

  const renderGame = () => {
    switch (activity.template) {
      case 'drag_match':
        return <DragMatchGame activity={activity} onResult={handleResult} />
      case 'trace_path':
        return <TracePathGame activity={activity} onResult={handleResult} />
      case 'listen_identify':
        return <ListenIdentifyGame activity={activity} onResult={handleResult} />
      default:
        return <p>القالب غير مدعوم حالياً</p>
    }
  }

  return (
    <div className="activity-screen" dir="rtl">
      <button onClick={onExit} className="btn-back" aria-label="عودة">
        ← خروج
      </button>
      
      <div className="activity-content">
        {renderGame()}
      </div>
    </div>
  )
}

## 📋 Anti-patterns الممنوعة في P3
