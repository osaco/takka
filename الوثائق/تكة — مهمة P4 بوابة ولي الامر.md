### Prompt 2 — تنفيذ Dexie

```
أنشئ تنفيذ Dexie في src/data/progress-store.ts

```typescript
import Dexie, type Table from 'dexie'
import type { ProgressRepository } from '@domain/storage/repository'
import type { ChildProfile, Evidence, SessionRecord } from '@domain/types'

class TakkaDB extends Dexie {
  profiles!: Table<ChildProfile, string>
  evidences!: Table<Evidence & { profileId: string }, number>
  sessions!: Table<SessionRecord & { profileId: string }, number>

  constructor() {
    super('takkaDB')
    this.version(1).stores({
      profiles: 'id, createdAt, updatedAt',
      evidences: '++, profileId, skillId, at',
      sessions: '++, profileId, startedAt',
    })
  }
}

const db = new TakkaDB()

export class DexieProgressStore implements ProgressRepository {
  async loadProfile(profileId: string): Promise<ChildProfile | null> {
    return db.profiles.get(profileId) ?? null
  }

  async saveProfile(profile: ChildProfile): Promise<void> {
    profile.updatedAt = Date.now()
    await db.profiles.put(profile)
  }

  async saveEvidence(evidence: Evidence): Promise<void> {
    await db.evidences.add({ ...evidence, profileId: '' })
  }

  async saveSession(session: SessionRecord): Promise<void> {
    await db.sessions.add(session)
  }

  async getSkillStates(profileId: string): Promise<Record<string, SkillState>> {
    const evidences = await db.evidences.where('profileId').equals(profileId).toArray()
    const states: Record<string, SkillState> = {}
    
    for (const evidence of evidences) {
      const currentState = states[evidence.skillId] || 'not_reached'
      // Simplified state computation
      if (evidence.kind === 'independent') {
        states[evidence.skillId] = 'independent'
      } else if (currentState !== 'independent') {
        states[evidence.skillId] = 'with_help'
      }
    }
    
    return states
  }

  async getLastSession(profileId: string): Promise<SessionRecord | null> {
    const sessions = await db.sessions
      .where('profileId')
      .equals(profileId)
      .reverse()
      .toArray()
    return sessions[0] ?? null
  }

  async exportProfile(profileId: string): Promise<Blob> {
    const profile = await this.loadProfile(profileId)
    const evidences = await db.evidences.where('profileId').equals(profileId).toArray()
    const sessions = await db.sessions.where('profileId').equals(profileId).toArray()
    
    const data = {
      version: '1.0',
      exportedAt: Date.now(),
      profile,
      evidences,
      sessions,
      checksum: this.computeChecksum(profileId, evidences.length),
    }
    
    return new Blob([JSON.stringify(data, null, 2)], { type: 'application/json' })
  }

  async importProfile(file: Blob): Promise<ChildProfile & { evidences: Evidence[] }> {
    const text = await file.text()
    const data = JSON.parse(text)
    
    if (data.version !== '1.0') {
      throw new Error('نسخة غير مدعومة')
    }
    
    // Verify checksum (simplified)
    const expectedChecksum = this.computeChecksum(data.profile.id, data.evidences.length)
    if (data.checksum !== expectedChecksum) {
      throw new Error('ملف تالف — فشل فحص السلامة')
    }
    
    await this.saveProfile(data.profile)
    for (const evidence of data.evidences) {
      await db.evidences.add({ ...evidence, profileId: data.profile.id })
    }
    
    return data
  }

  async deleteProfile(profileId: string): Promise<void> {
    await db.evidences.where('profileId').equals(profileId).delete()
    await db.sessions.where('profileId').equals(profileId).delete()
    await db.profiles.delete(profileId)
  }

  async getStatistics(profileId: string) {
    const evidences = await db.evidences.where('profileId').equals(profileId).toArray()
    const sessions = await db.sessions.where('profileId').equals(profileId).toArray()
    
    const independentCount = evidences.filter(e => e.kind === 'independent').length
    const skillsWithHelp = new Set(
      evidences.filter(e => e.kind === 'with_hint' || e.kind === 'with_scaffold')
        .map(e => e.skillId)
    ).size
    
    return {
      totalSessions: sessions.length,
      totalEvidence: evidences.length,
      independentCount,
      skillsWithHelp,
      avgAttemptsPerSkill: evidences.length > 0
        ? evidences.reduce((sum, e) => sum + e.attempts, 0) / new Set(evidences.map(e => e.skillId)).size
        : 0,
    }
  }

  private computeChecksum(profileId: string, evidenceCount: number): string {
    return btoa(`${profileId}:${evidenceCount}:${Date.now()}`)
  }
}

export const progressStore = new DexieProgressStore()

```tsx
import React, { useState, useEffect } from 'react'
import { useAppStore } from '../store/app-store'
import { progressStore } from '../../data/progress-store'
import type { ChildProfile } from '@domain/types'
import './parent-gate.css'

export const ParentGateScreen: React.FC = () => {
  const { isSoundEnabled, toggleSound, setIsLowPerformanceMode } = useAppStore()
  const [profiles, setProfiles] = useState<ChildProfile[]>([])
  const [selectedProfile, setSelectedProfile] = useState<string | null>(null)
  const [showAdd, setShowAdd] = useState(false)
  const [newName, setNewName] = useState('')
  const [newGrade, setNewGrade] = useState<1 | 2 | 3 | 4 | 5 | 6>(1)

  useEffect(() => {
    loadProfiles()
  }, [])

  const loadProfiles = async () => {
    const all = await (progressStore as any).getAllProfiles?.() || []
    setProfiles(all)
  }

  const handleAddProfile = async () => {
    if (!newName.trim()) return
    const profile: ChildProfile = {
      id: `profile-${Date.now()}`,
      displayName: newName.trim(),
      grade: newGrade,
      createdAt: Date.now(),
      updatedAt: Date.now(),
    }
    await progressStore.saveProfile(profile)
    setProfiles(prev => [...prev, profile])
    setSelectedProfile(profile.id)
    setShowAdd(false)
    setNewName('')
  }

  const handleExport = async () => {
    if (!selectedProfile) return
    const blob = await progressStore.exportProfile(selectedProfile)
    const url = URL.createObjectURL(blob)
    const a = document.createElement('a')
    a.href = url
    a.download = `تكة-تقدم-${newName || 'طفل'}.json`
    a.click()
    URL.revokeObjectURL(url)
  }

  const handleImport = async (e: React.ChangeEvent<HTMLInputElement>) => {
    const file = e.target.files?.[0]
    if (!file) return
    try {
      const data = await progressStore.importProfile(file)
      setProfiles(prev => [...prev, data])
      setSelectedProfile(data.id)
    } catch (err) {
      alert('فشل الاستيراد: ' + (err instanceof Error ? err.message : String(err)))
    }
  }

  return (
    <div className="parent-gate" dir="rtl">
      <h1>بوابة ولي الأمر</h1>
      
      <section className="profiles-section">
        <h2>ملفات الأطفال</h2>
        {profiles.map(p => (
          <button
            key={p.id}
            className={`profile-card ${selectedProfile === p.id ? 'selected' : ''}`}
            onClick={() => setSelectedProfile(p.id)}
          >
            <span className="profile-name">{p.displayName}</span>
            <span className="profile-grade">الصف {p.grade}</span>
          </button>
        ))}
        
        <button className="btn-add" onClick={() => setShowAdd(true)}>
          + إضافة طفل جديد
        </button>
      </section>

      {showAdd && (
        <div className="add-profile-modal">
          <h3>إضافة طفل جديد</h3>
          <input
            type="text"
            placeholder="الاسم المستعار"
            value={newName}
            onChange={e => setNewName(e.target.value)}
            maxLength={20}
          />
          <select value={newGrade} onChange={e => setNewGrade(Number(e.target.value) as any)}>
            {[1,2,3,4,5,6].map(g => <option key={g} value={g}>الصف {g}</option>)}
          </select>
          <div className="modal-actions">
            <button onClick={handleAddProfile}>حفظ</button>
            <button onClick={() => setShowAdd(false)}>إلغاء</button>
          </div>
        </div>
      )}

      {selectedProfile && (
        <section className="actions-section">
          <h2>الإجراءات</h2>
          
          <div className="setting-row">
            <span>الصوت</span>
            <button onClick={toggleSound} className={`toggle ${isSoundEnabled ? 'on' : 'off'}`}>
              {isSoundEnabled ? 'مفعّل' : 'معطّل'}
            </button>
          </div>

          <div className="setting-row">
            <span>الوضع منخفض الأداء</span>
            <button 
              onClick={() => setIsLowPerformanceMode(!useAppStore.getState().isLowPerformanceMode)}
              className={`toggle ${useAppStore.getState().isLowPerformanceMode ? 'on' : 'off'}`}
            >
              {useAppStore.getState().isLowPerformanceMode ? 'مفعّل' : 'معطّل'}
            </button>
          </div>

          <div className="export-import">
            <button onClick={handleExport} className="btn-export">
              📤 تصدير التقدم
            </button>
            <label className="btn-import">
              📥 استيراد تقدم
              <input type="file" accept=".json" onChange={handleImport} hidden />
            </label>
          </div>

          <p className="privacy-note">
            جميع البيانات محفوظة محلياً على الجهاز. لا تُرفع إلى أي خادم.
          </p>
        </section>
      )}

      <button className="btn-back" onClick={() => useAppStore.getState().setCurrentScreen('world-map')}>
        ← العودة للتطبيق
      </button>
    </div>
  )
}

### Prompt 4 — Progress Screen

```
أنشئ شاشة عرض التقدّم في src/runtime/screens/progress-screen.tsx

```tsx
import React, { useEffect, useState } from 'react'
import { progressStore } from '../../data/progress-store'
import type { ChildProfile, Record<string, SkillState> } from '@domain/types'

interface ProgressScreenProps {
  profileId: string
  onBack: () => void
}

export const ProgressScreen: React.FC<ProgressScreenProps> = ({ profileId, onBack }) => {
  const [profile, setProfile] = useState<ChildProfile | null>(null)
  const [skillStates, setSkillStates] = useState<Record<string, SkillState>>({})
  const [stats, setStats] = useState<any>(null)

  useEffect(() => {
    loadData()
  }, [profileId])

  const loadData = async () => {
    const p = await progressStore.loadProfile(profileId)
    const states = await progressStore.getSkillStates(profileId)
    const s = await progressStore.getStatistics(profileId)
    setProfile(p)
    setSkillStates(states)
    setStats(s)
  }

  if (!profile) return <p>جارٍ التحميل...</p>

  const independentCount = Object.values(skillStates).filter(s => s === 'independent').length
  const totalCount = Object.keys(skillStates).length
  const coverage = totalCount > 0 ? Math.round((independentCount / totalCount) * 100) : 0

  return (
    <div className="progress-screen" dir="rtl">
      <button onClick={onBack} className="btn-back">← عودة</button>
      
      <h1>تقدّم {profile.displayName}</h1>
      
      <div className="stats-grid">
        <div className="stat-card">
          <span className="stat-value">{coverage}%</span>
          <span className="stat-label">نسبة الإتقان</span>
        </div>
        <div className="stat-card">
          <span className="stat-value">{stats?.totalSessions ?? 0}</span>
          <span className="stat-label">الجلسات</span>
        </div>
        <div className="stat-card">
          <span className="stat-value">{stats?.independentCount ?? 0}</span>
          <span className="stat-label">مهارات متقنة</span>
        </div>
        <div className="stat-card">
          <span className="stat-value">{stats?.skillsWithHelp ?? 0}</span>
          <span className="stat-label">تحتاج مراجعة</span>
        </div>
      </div>

      <h2>حالات المهارات</h2>
      <div className="skills-list">
        {Object.entries(skillStates).map(([skillId, state]) => (
          <div key={skillId} className={`skill-row ${state}`}>
            <span className="skill-id">{skillId}</span>
            <span className="skill-state">
              {state === 'independent' && '✓ متقنة'}
              {state === 'with_help' && '◐ بمساعدة'}
              {state === 'tried' && '○ جرّب'}
              {state === 'not_reached' && '○ لم يصل'}
            </span>
          </div>
        ))}
      </div>

      <p className="privacy-notice">
        هذه البيانات مخزنة محلياً فقط. استخدم «تصدير التقدم» للنسخ الاحتياطي.
      </p>
    </div>
  )
}

## 📋 Anti-patterns الممنوعة في P4

| ممنوع | البديل |
|:---|:---|
| تخزين أسماء حقيقية | أسماء مستعارة فقط |
| إرسال بيانات للشبكة | محلي 100% |
| كشف هوية الطفل | لا هوية على الإطلاق |
